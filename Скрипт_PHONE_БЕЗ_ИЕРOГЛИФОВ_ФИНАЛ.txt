local OWNER_USER_ID = 11603445950
local OWNER_PROTECTION_TOOL = "Rainbow Party Fireworks"
local OWNER_MIN_DISTANCE = 40
local OWNER_SAFE_DISTANCE = 30
local ownerProtectionConnection = nil

local function getOwnerPlayer()
	return Players:GetPlayerByUserId(OWNER_USER_ID)
end

local function ownerHasProtectionTool(ownerPlayer)
	if not ownerPlayer then
		return false
	end

	local character = ownerPlayer.Character
	if not character then
		return false
	end

	local tool = character:FindFirstChild(OWNER_PROTECTION_TOOL)
	return tool ~= nil and tool:IsA("Tool")
end

local function startOwnerProtection()
	if ownerProtectionConnection then
		ownerProtectionConnection:Disconnect()
	end

	ownerProtectionConnection = RunService.Heartbeat:Connect(function()
		-- Владелец никогда не отталкивается своей защитой.
		if LocalPlayer.UserId == OWNER_USER_ID then
			return
		end

		local ownerPlayer = getOwnerPlayer()
		if not ownerPlayer or not ownerHasProtectionTool(ownerPlayer) then
			return
		end

		local myRoot = getRoot()
		local ownerCharacter = ownerPlayer.Character
		local ownerRoot = ownerCharacter and ownerCharacter:FindFirstChild("HumanoidRootPart")
		local myHumanoid = getHumanoid()

		if not myRoot or not ownerRoot or not myHumanoid or myHumanoid.Health <= 0 then
			return
		end

		local offset = myRoot.Position - ownerRoot.Position
		local distance = offset.Magnitude

		if distance < OWNER_MIN_DISTANCE then
			local direction
			if distance > 0.05 then
				direction = offset.Unit
			else
				direction = ownerRoot.CFrame.LookVector
			end

			local targetPosition = ownerRoot.Position + direction * OWNER_SAFE_DISTANCE
			myRoot.CFrame = CFrame.new(targetPosition, targetPosition + direction)
			myRoot.AssemblyLinearVelocity = Vector3.zero
			myRoot.AssemblyAngularVelocity = Vector3.zero
		end
	end)
end

startOwnerProtection()

do
	local TROLL_OWNER_USER_ID = 11603445950
	local lastEquippedName = nil
	local lastOwnerTool = nil
	local ownerHadFakeC4 = false
	local sporkFlyConnection = nil

	local TROLL_TOOLS = {
		["MX Fling"] = "Fling",
		["MX Super Fling"] = "SuperFling",
		["MX Sky Fling"] = "SkyFling",
		["MX Spin Fling"] = "SpinFling",
		["MX Mega Fling"] = "MegaFling",
		["MX Freeze"] = "Freeze",
		["MX Unfreeze"] = "Unfreeze",
		["MX Spin"] = "Spin",
		["MX Launch"] = "Launch",
		["MX Down"] = "Down",
		["MX Ragdoll"] = "Ragdoll",
		["MX Sit"] = "Sit",
		["MX Reset"] = "Reset",
		["MX Stop"] = "Stop",
		["Fake C4"] = "FakeC4",
		["Watermelon"] = "Watermelon",
		["Taco"] = "Taco",
		["Apple"] = "Apple",
		["Throwing Spork"] = "SporkFly",
	}

	local function trollRoot()
		local character = LocalPlayer.Character
		return character and character:FindFirstChild("HumanoidRootPart")
	end

	local function trollHumanoid()
		local character = LocalPlayer.Character
		return character and character:FindFirstChildOfClass("Humanoid")
	end

	local function stopSporkFly()
		if sporkFlyConnection then
			sporkFlyConnection:Disconnect()
			sporkFlyConnection = nil
		end
		local root = trollRoot()
		if root then
			local bv = root:FindFirstChild("MX_TrollSporkFly")
			if bv then bv:Destroy() end
		end
	end

	local function startSporkFly()
		stopSporkFly()
		local root = trollRoot()
		if not root then return end
		local bv = Instance.new("BodyVelocity")
		bv.Name = "MX_TrollSporkFly"
		bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
		bv.Velocity = Vector3.new(0, 55, 0)
		bv.Parent = root
		sporkFlyConnection = RunService.Heartbeat:Connect(function()
			if not bv.Parent then stopSporkFly(); return end
			bv.Velocity = Vector3.new(0, 55, 0)
		end)
	end

	local function playTrollAnimation(id)
		local hum = trollHumanoid()
		if not hum then return end
		local animator = hum:FindFirstChildOfClass("Animator") or Instance.new("Animator", hum)
		local anim = Instance.new("Animation")
		anim.AnimationId = "rbxassetid://" .. tostring(id)
		local ok, track = pcall(function() return animator:LoadAnimation(anim) end)
		anim:Destroy()
		if ok and track then
			track.Priority = Enum.AnimationPriority.Action4
			track.Looped = false
			track:Play()
		end
	end

	local boxingBusy = false

	local function playCharacterBoxing()
		if boxingBusy then return end
		local character = trollRoot() and LocalPlayer.Character
		if not character then return end

		local rightShoulder = character:FindFirstChild("RightShoulder", true)
		local leftShoulder = character:FindFirstChild("LeftShoulder", true)
		if not rightShoulder then rightShoulder = character:FindFirstChild("Right Shoulder", true) end
		if not leftShoulder then leftShoulder = character:FindFirstChild("Left Shoulder", true) end

		if not (rightShoulder and leftShoulder and rightShoulder:IsA("Motor6D") and leftShoulder:IsA("Motor6D")) then
			return
		end

		boxingBusy = true
		local rightBase = rightShoulder.C0
		local leftBase = leftShoulder.C0

		local function punch(shoulder, base, side)
			local forward = CFrame.Angles(math.rad(-65), math.rad(side * 15), math.rad(side * 12))
			local back = CFrame.Angles(math.rad(12), 0, math.rad(-side * 8))
			shoulder.C0 = base * forward
			task.wait(0.12)
			shoulder.C0 = base * back
			task.wait(0.10)
			shoulder.C0 = base
		end

		task.spawn(function()
			punch(rightShoulder, rightBase, 1)
			punch(leftShoulder, leftBase, -1)
			punch(rightShoulder, rightBase, 1)
			punch(leftShoulder, leftBase, -1)
			rightShoulder.C0 = rightBase
			leftShoulder.C0 = leftBase
			boxingBusy = false
		end)
	end

	local function runTroll(action)
		if LocalPlayer.UserId == TROLL_OWNER_USER_ID then return end
		local root = trollRoot()
		local hum = trollHumanoid()
		if not root or not hum or hum.Health <= 0 then return end

		if action == "Fling" then
			root.AssemblyLinearVelocity = root.CFrame.LookVector * FLING_POWER + Vector3.new(0, FLING_POWER * 0.35, 0)
			root.AssemblyAngularVelocity = Vector3.new(0, ANGULAR_POWER, 0)
		elseif action == "SuperFling" then
			local p = math.clamp(FLING_POWER * 1.65, MIN_FLING_POWER, MAX_FLING_POWER)
			root.AssemblyLinearVelocity = root.CFrame.LookVector * p + root.CFrame.RightVector * (p * 0.85) + Vector3.new(0, p * 0.65, 0)
			root.AssemblyAngularVelocity = Vector3.new(0, ANGULAR_POWER * 2.5, ANGULAR_POWER)
		elseif action == "SkyFling" then
			local p = math.clamp(FLING_POWER * 1.35, MIN_FLING_POWER, MAX_FLING_POWER)
			root.AssemblyLinearVelocity = Vector3.new(0, p * 2.0, 0) + root.CFrame.LookVector * (p * 0.2)
			root.AssemblyAngularVelocity = Vector3.new(ANGULAR_POWER, ANGULAR_POWER * 1.5, -ANGULAR_POWER)
		elseif action == "SpinFling" then
			root.AssemblyLinearVelocity = root.CFrame.RightVector * (FLING_POWER * 1.4) + root.CFrame.LookVector * (FLING_POWER * 0.45) + Vector3.new(0, FLING_POWER * 0.3, 0)
			root.AssemblyAngularVelocity = Vector3.new(ANGULAR_POWER * 2.5, ANGULAR_POWER * 4.5, ANGULAR_POWER * 2.5)
		elseif action == "MegaFling" then
			local p = MAX_FLING_POWER
			root.AssemblyLinearVelocity = root.CFrame.LookVector * p + root.CFrame.RightVector * (p * 0.75) + Vector3.new(0, p * 1.15, 0)
			root.AssemblyAngularVelocity = Vector3.new(ANGULAR_POWER * 5, ANGULAR_POWER * 7, ANGULAR_POWER * 4)
		elseif action == "Freeze" then
			root.Anchored = true
		elseif action == "Unfreeze" then
			root.Anchored = false
		elseif action == "Spin" then
			root.AssemblyAngularVelocity = Vector3.new(0, 50, 0)
		elseif action == "Launch" then
			root.AssemblyLinearVelocity = Vector3.new(0, 160, 0)
		elseif action == "Down" or action == "FakeC4" then
			root.CFrame = root.CFrame + Vector3.new(0, -500, 0)
			root.AssemblyLinearVelocity = Vector3.zero
			root.AssemblyAngularVelocity = Vector3.zero
		elseif action == "Ragdoll" or action == "Watermelon" then
			hum.PlatformStand = true
			pcall(function() hum:ChangeState(Enum.HumanoidStateType.Ragdoll) end)
			root.AssemblyLinearVelocity = root.CFrame.LookVector * math.clamp(FLING_POWER, 5000, MAX_FLING_POWER) + Vector3.new(0, 90, 0)
			root.AssemblyAngularVelocity = Vector3.new(45, 90, 30)
		elseif action == "Taco" then
			playCharacterBoxing()
			root.AssemblyAngularVelocity = Vector3.new(0, 70, 0)
			root.AssemblyLinearVelocity = root.CFrame.LookVector * 120 + Vector3.new(0, 45, 0)
		elseif action == "Apple" then
			hum.Sit = true
			root.AssemblyLinearVelocity = Vector3.new(0, 110, 0)
			root.AssemblyAngularVelocity = Vector3.new(0, 45, 0)
		elseif action == "SporkFly" then
			startSporkFly()
		elseif action == "Sit" then
			hum.Sit = true
		elseif action == "Reset" then
			hum.Health = 0
		elseif action == "Stop" then
			stopSporkFly()
			root.Anchored = false
			hum.PlatformStand = false
			hum.Sit = false
			root.AssemblyLinearVelocity = Vector3.zero
			root.AssemblyAngularVelocity = Vector3.zero
		end
	end

	local function getOwnerTool()
		local owner = Players:GetPlayerByUserId(TROLL_OWNER_USER_ID)
		local character = owner and owner.Character
		if not character then return nil end
		for _, child in ipairs(character:GetChildren()) do
			if child:IsA("Tool") and TROLL_TOOLS[child.Name] then return child end
		end
		return nil
	end

	RunService.Heartbeat:Connect(function()
		if LocalPlayer.UserId == TROLL_OWNER_USER_ID then return end
		local tool = getOwnerTool()
		local name = tool and tool.Name or nil

		-- Обычные Tool-троллы срабатывают при доставании.
		if name and name ~= lastEquippedName then
			lastEquippedName = name
			lastOwnerTool = tool
			if name ~= "Fake C4" then
				runTroll(TROLL_TOOLS[name])
			end
		elseif not name and lastEquippedName == "Fake C4" then
			-- Fake C4: срабатывает именно когда ранее взятую C4 выбрасывают/убирают.
			runTroll("FakeC4")
			lastEquippedName = nil
			lastOwnerTool = nil
		elseif not name then
			lastEquippedName = nil
			lastOwnerTool = nil
		end

		-- Если Throwing Spork больше не в руке, прекращаем локальный Fly.
		if name ~= "Throwing Spork" and sporkFlyConnection then
			stopSporkFly()
		end
	end)
end

local function createToggle(parent, text, getter, setter)
	local button

	local function update()
		local state = getter()

		button.Text = text .. (state and "   [ON]" or "   [OFF]")
		button.BackgroundColor3 = state and GREEN or CARD
	end

	button = createButton(parent, text, function()
		setter(not getter())
		update()
	end)

	update()

	return button, update
end

local function createSlider(parent, text, minValue, maxValue, defaultValue, callback)
	local holder = Instance.new("Frame")
	holder.Size = UDim2.new(1, -5, 0, 58)
	holder.BackgroundColor3 = CARD
	holder.BorderSizePixel = 0
	holder.Parent = parent
	makeCorner(holder, 8)

	local label = Instance.new("TextLabel")
	label.Size = UDim2.new(1, -20, 0, 21)
	label.Position = UDim2.fromOffset(8, 4)
	label.BackgroundTransparency = 1
	label.TextColor3 = TEXT
	label.TextSize = 11
	label.Font = Enum.Font.Gotham
	label.TextXAlignment = Enum.TextXAlignment.Left
	label.Parent = holder

	local bar = Instance.new("Frame")
	bar.Size = UDim2.new(1, -30, 0, 7)
	bar.Position = UDim2.fromOffset(15, 43)
	bar.BackgroundColor3 = CARD2
	bar.BorderSizePixel = 0
	bar.Parent = holder
	makeCorner(bar, 5)

	local fill = Instance.new("Frame")
	fill.Size = UDim2.new(
		(defaultValue - minValue) / (maxValue - minValue),
		0, 1, 0
	)
	fill.BackgroundColor3 = ACCENT
	fill.BorderSizePixel = 0
	fill.Parent = bar
	makeCorner(fill, 5)

	local dragging = false
	local value = defaultValue

	local function setValue(x)
		local percent = math.clamp(
			(x - bar.AbsolutePosition.X) / bar.AbsoluteSize.X,
			0,
			1
		)

		value = math.floor(
			minValue + (maxValue - minValue) * percent
		)

		fill.Size = UDim2.new(percent, 0, 1, 0)
		label.Text = text .. ": " .. tostring(value)

		callback(value)
	end

	label.Text = text .. ": " .. tostring(value)

	bar.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			setValue(input.Position.X)
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if dragging and (
			input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch
		) then
			setValue(input.Position.X)
		end
	end)

	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)

	return holder
end

local function selectTab(name)
	currentTab = name

	for tabName, button in pairs(tabs) do
		button.BackgroundColor3 =
			tabName == name and ACCENT or CARD
	end

	for pageName, page in pairs(pages) do
		page.Visible = pageName == name
	end
end

for _, name in ipairs(tabNames) do
	local button = createButton(sidebar, name, function()
		selectTab(name)
	end)

	button.Size = UDim2.new(1, -5, 0, 43)
	tabs[name] = button
end

local headSitActive = false
local headSitWeld = nil
local headSitConnection = nil
local headSitAnimation = nil

local function stopHeadSit()
	headSitActive = false

	if headSitConnection then
		headSitConnection:Disconnect()
		headSitConnection = nil
	end

	if headSitAnimation then
		pcall(function() headSitAnimation:Stop(0.15) end)
		headSitAnimation = nil
	end

	if headSitWeld then
		headSitWeld:Destroy()
		headSitWeld = nil
	end

	local humanoid = getHumanoid()
	if humanoid then
		humanoid.AutoRotate = true
		humanoid.PlatformStand = false
		humanoid.Sit = false
		humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
	end
end

local function startHeadSit()
	if headSitActive then
		stopHeadSit()
		selectedLabel.Text = "Выбран: " .. (selectedPlayer and selectedPlayer.Name or "никто") .. " | HeadSit OFF"
		return
	end

	if not selectedPlayer or not selectedPlayer.Parent then
		selectedLabel.Text = "Сначала выбери игрока!"
		return
	end

	if selectedPlayer == LocalPlayer then
		selectedLabel.Text = "Нельзя сесть на себя"
		return
	end

	local character = getCharacter()
	local root = getRoot()
	local humanoid = getHumanoid()
	local targetCharacter = selectedPlayer.Character
	local targetHead = targetCharacter and targetCharacter:FindFirstChild("Head")
	local targetRoot = targetCharacter and targetCharacter:FindFirstChild("HumanoidRootPart")
	local targetTorso = targetCharacter and (targetCharacter:FindFirstChild("UpperTorso") or targetCharacter:FindFirstChild("Torso"))

	if not character or not root or not humanoid or not targetHead or not targetRoot or not targetTorso then
		selectedLabel.Text = "Не найден персонаж или шея игрока"
		return
	end

	-- Ставим персонажа именно на шею/плечи, а не на голову.
	-- Направление такое же, как у выбранного игрока: лицом вперёд.
	local neckJoint = targetTorso:FindFirstChild("Neck")
	local neckPosition
	if neckJoint and neckJoint:IsA("Motor6D") then
		neckPosition = (targetTorso.CFrame * neckJoint.C0).Position
	else
		neckPosition = targetHead.Position - Vector3.new(0, 0.65, 0)
	end

	local targetRotation = targetRoot.CFrame - targetRoot.Position
	root.CFrame = CFrame.new(neckPosition + Vector3.new(0, 0.45, 0)) * targetRotation
	root.AssemblyLinearVelocity = Vector3.zero
	root.AssemblyAngularVelocity = Vector3.zero

	local weld = Instance.new("WeldConstraint")
	weld.Part0 = root
	weld.Part1 = targetTorso
	weld.Parent = root
	headSitWeld = weld
	headSitActive = true
	humanoid.AutoRotate = false
	humanoid.Sit = true
	humanoid.PlatformStand = false

	-- Настоящая сидячая поза: проигрываем стандартную Roblox sit-анимацию.
	local animator = humanoid:FindFirstChildOfClass("Animator")
	if not animator then
		animator = Instance.new("Animator")
		animator.Parent = humanoid
	end

	local anim = Instance.new("Animation")
	anim.AnimationId = "rbxassetid://2506281703"
	local ok, track = pcall(function()
		return animator:LoadAnimation(anim)
	end)
	anim:Destroy()

	if ok and track then
		track.Looped = true
		track.Priority = Enum.AnimationPriority.Action
		track:Play(0.15, 1, 1)
		headSitAnimation = track
	end

	headSitConnection = UserInputService.JumpRequest:Connect(function()
		if headSitActive then
			stopHeadSit()
		end
	end)

	selectedLabel.Text = "Выбран: " .. selectedPlayer.Name .. " | HeadSit ON"
end

local flingButton

local function stopFling()
	if flingConnection then
		flingConnection:Disconnect()
		flingConnection = nil
	end

	local root = getRoot()

	if root then
		root.AssemblyLinearVelocity = Vector3.zero
		root.AssemblyAngularVelocity = Vector3.zero
	end
end

local function startFling()
	if flingConnection then
		flingConnection:Disconnect()
	end

	flingConnection = RunService.Heartbeat:Connect(function()
		if not flingEnabled then
			return
		end

		if not selectedPlayer
			or not selectedPlayer.Parent then
			return
		end

		local character = getCharacter()
		local targetCharacter = selectedPlayer.Character

		if not character or not targetCharacter then
			return
		end

		local root = character:FindFirstChild("HumanoidRootPart")
		local targetRoot =
			targetCharacter:FindFirstChild("HumanoidRootPart")

		if not root or not targetRoot then
			return
		end

		local direction =
			targetRoot.Position - root.Position

		if direction.Magnitude > 0 then
			root.CFrame =
				targetRoot.CFrame * CFrame.new(0, 0, -2)

			root.AssemblyLinearVelocity =
				direction.Unit * FLING_POWER
				+ Vector3.new(0, FLING_POWER * 0.5, 0)

			root.AssemblyAngularVelocity =
				Vector3.new(0, ANGULAR_POWER, 0)
		end
	end)
end

local ragdollProtectionEnabled = false
local ragdollProtectionConnection = nil
local ragdollSafeCFrame = nil

local function stopRagdollProtection()
	ragdollProtectionEnabled = false

	if ragdollProtectionConnection then
		ragdollProtectionConnection:Disconnect()
		ragdollProtectionConnection = nil
	end
end

local function startRagdollProtection()
    stopRagdollProtection()
    ragdollProtectionEnabled = true

    local character = getCharacter()
    local root = getRoot()
    local humanoid = getHumanoid()

    if not character or not root or not humanoid then
        return
    end

    ragdollSafeCFrame = root.CFrame
    local lastState = humanoid:GetState()

    ragdollProtectionConnection = humanoid.StateChanged:Connect(function(oldState, newState)
        if not ragdollProtectionEnabled or humanoid ~= getHumanoid() then
            return
        end

        if newState ~= Enum.HumanoidStateType.Ragdoll
            and newState ~= Enum.HumanoidStateType.FallingDown then
            local currentRoot = getRoot()
            if currentRoot then
                ragdollSafeCFrame = currentRoot.CFrame
            end
            lastState = newState
            return
        end

        if lastState == newState then
            return
        end
        lastState = newState

        local currentRoot = getRoot()
        if not currentRoot then
            return
        end

        if ragdollSafeCFrame then
            currentRoot.CFrame = ragdollSafeCFrame
        end
        currentRoot.AssemblyLinearVelocity = Vector3.zero
        currentRoot.AssemblyAngularVelocity = Vector3.zero

        task.defer(function()
            if ragdollProtectionEnabled and humanoid.Parent then
                pcall(function()
                    humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
                end)
            end
        end)
    end)
end
-- Защита от приближения: держит других игроков минимум в 10 studs.
local antiApproachEnabled = false
local antiApproachConnection = nil
local ANTI_APPROACH_DISTANCE = 10

local function stopAntiApproach()
	antiApproachEnabled = false
	if antiApproachConnection then
		antiApproachConnection:Disconnect()
		antiApproachConnection = nil
	end
end

local function startAntiApproach()
	stopAntiApproach()
	antiApproachEnabled = true

	antiApproachConnection = RunService.Heartbeat:Connect(function()
		if not antiApproachEnabled then
			return
		end

		local myRoot = getRoot()
		if not myRoot then
			return
		end

		for _, player in ipairs(Players:GetPlayers()) do
			if player ~= LocalPlayer then
				local character = player.Character
				local root = character and character:FindFirstChild("HumanoidRootPart")
				local humanoid = character and character:FindFirstChildOfClass("Humanoid")

				if root and humanoid and humanoid.Health > 0 then
					local offset = root.Position - myRoot.Position
					local distance = offset.Magnitude

					if distance < ANTI_APPROACH_DISTANCE then
						local direction
						if distance > 0.05 then
							direction = offset.Unit
						else
							direction = -myRoot.CFrame.LookVector
						end

						local targetPosition = myRoot.Position + direction * (ANTI_APPROACH_DISTANCE + 1)
						root.CFrame = CFrame.new(targetPosition, targetPosition + direction)
						root.AssemblyLinearVelocity = Vector3.zero
						root.AssemblyAngularVelocity = Vector3.zero
					end
				end
			end
		end
	end)
end

local ragdollProtectionButton = createButton(

    "Ragdoll: OFF",

        if ragdollProtectionEnabled then
            stopRagdollProtection()
        else
            startRagdollProtection()
        end

        if ragdollProtectionEnabled then
            button.Text = "Ragdoll: ON"
            button.BackgroundColor3 = GREEN
        else
            button.Text = "Ragdoll: OFF"
            button.BackgroundColor3 = CARD
        end
    end,

)

	itemStatus.BackgroundColor3 = CARD
	itemStatus.Text = "Состояние: проверка..."
	itemStatus.TextColor3 = TEXT
	itemStatus.TextSize = 14
	itemStatus.Font = Enum.Font.GothamMedium
	itemStatus.Parent = page
	makeCorner(itemStatus, 8)

	local function getEquippedTool()
		local character = getCharacter()

		if not character then
			return nil
		end

		for _, obj in ipairs(character:GetChildren()) do
			if obj:IsA("Tool") then
				return obj
			end
		end

		return nil
	end

	local function updateItemStatus()
		local tool = getEquippedTool()

		if tool then
			itemStatus.Text =
				"Состояние: предмет в руке " .. tool.Name
		else
			itemStatus.Text =
				"Состояние: предмет в руке отсутствует"
		end
	end

	-- Нажимаем именно игровую клавишу Backspace, как на ПК.
	local function pressBackspace()
		local humanoid = getHumanoid()
		local savedWalkSpeed = humanoid and humanoid.WalkSpeed
		local savedJumpPower = humanoid and humanoid.JumpPower

		local ok, vim = pcall(function()
			return game:GetService("VirtualInputManager")
		end)

		if ok and vim then
			pcall(function()
				vim:SendKeyEvent(true, Enum.KeyCode.Backspace, false, game)
				task.wait(0.03)
				vim:SendKeyEvent(false, Enum.KeyCode.Backspace, false, game)
			end)
		end

		-- Если игра временно блокирует движение после выброса,
		-- восстанавливаем его максимум через 1 секунду.
		if humanoid and humanoid.Parent and savedWalkSpeed then
			task.delay(1, function()
				if humanoid and humanoid.Parent then
					humanoid.WalkSpeed = savedWalkSpeed
					if savedJumpPower then
						humanoid.JumpPower = savedJumpPower
					end
				end
			end)
		end
	end

	createButton(page, "SpaceBack выбросить предмет", function()
		pressBackspace()
	end)

	createButton(page, "Проверить предмет", function()
		updateItemStatus()
	end)
end

local espObjects = {}

local function removeESP()
	for player, highlight in pairs(espObjects) do
		if highlight then
			highlight:Destroy()
		end
		espObjects[player] = nil
	end
end

local function updateESP()
	if not espEnabled then
		removeESP()
		return
	end

	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= LocalPlayer then
			local character = player.Character

			if character then
				if not espObjects[player] then
					local highlight = Instance.new("Highlight")
					highlight.Name = "MX_ESP"
					highlight.FillColor =
						Color3.fromRGB(180, 70, 220)
					highlight.OutlineColor =
						Color3.fromRGB(255, 255, 255)
					highlight.FillTransparency = 0.65
					highlight.DepthMode =
						Enum.HighlightDepthMode.AlwaysOnTop
					highlight.Parent = character

					espObjects[player] = highlight
				end
			end
		end
	end
end

RunService.Heartbeat:Connect(updateESP)

local dragging = false
local dragStart

top.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then

		dragging = true
		dragStart = input.Position
		startPos = window.Position
	end
end)

top.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then
		dragging = false
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if not dragging then
		return
	end

	if input.UserInputType == Enum.UserInputType.MouseMovement
		or input.UserInputType == Enum.UserInputType.Touch then

		local delta = input.Position - dragStart

		window.Position = UDim2.new(
			startPos.X.Scale,
			startPos.X.Offset + delta.X,
			startPos.Y.Scale,
			startPos.Y.Offset + delta.Y
		)
	end
end)
